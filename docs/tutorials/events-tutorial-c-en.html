Basics - Signal Connection

What are GObject Signals?

GObject signals are the core of GTK/GLib event-driven programming:

  • Event notification: Automatically called when something happens to an object
  • Type safe: Type checking at compile time
  • Flexible: Multiple handlers can be connected to the same signal

BlinkGTK Signals: | Signal Name | Firing Timing | Usage | |-------------|--------------|-------| | load-changed | Load state changed | Progress display | | load-failed | Load failed | Error handling | | title-changed | Title changed | Window title update | | uri-changed | URI changed | Address bar update |

Your First Signal Handler

Let's start with the simplest example:

#include <gtk/gtk.h>
#include <blink_gtk/blink_gtk.h>

/* Signal handler */
static void on_load_changed(GtkWidget* web_view,
                            BlinkLoadEvent load_event,
                            gpointer user_data) {
    const char* event_name = "";

    switch (load_event) {
        case BLINK_LOAD_STARTED:
            event_name = "Started";
            break;
        case BLINK_LOAD_REDIRECTED:
            event_name = "Redirected";
            break;
        case BLINK_LOAD_COMMITTED:
            event_name = "Committed";
            break;
        case BLINK_LOAD_FINISHED:
            event_name = "Finished";
            break;
    }

    g_print("Event: %s\n", event_name);
}

static void on_activate(GtkApplication* app, gpointer user_data) {
    GtkWidget* window = gtk_application_window_new(app);
    GtkWidget* web_view = blink_web_view_new();

    /* Connect signal */
    g_signal_connect(web_view, "load-changed",
                     G_CALLBACK(on_load_changed), NULL);

    gtk_window_set_child(GTK_WINDOW(window), web_view);
    blink_web_view_load_uri(BLINK_WEB_VIEW(web_view), "https://example.com");
    gtk_window_present(GTK_WINDOW(window));
}

int main(int argc, char* argv[]) {
    blink_gtk_init(&argc, &argv);  /* GTK4 は引数なし */

    GtkApplication* app = gtk_application_new("org.example.signals", 0);
    g_signal_connect(app, "activate", G_CALLBACK(on_activate), NULL);

    /* ⚠️ g_application_run() は使用禁止 — Chromium RunLoop が飢餓状態になる
     * g_application_register/activate で GtkApplication シグナルを発火させてから
     * blink_gtk_run_main_loop() でChromium RunLoop を起動する (Build 658実証済み) */
    g_application_register(G_APPLICATION(app), NULL, NULL);
    g_application_activate(G_APPLICATION(app));
    int status = blink_gtk_run_main_loop();

    g_object_unref(app);
    blink_gtk_shutdown();
    return status;
}

Using user_data

How to pass additional data to signal handlers:

#include <gtk/gtk.h>
#include <blink_gtk/blink_gtk.h>

/* Application data */
typedef struct {
    GtkWidget* window;
    GtkWidget* status_label;
    int load_count;
} AppData;

/* Signal handler */
static void on_load_changed(GtkWidget* web_view,
                            BlinkLoadEvent load_event,
                            gpointer user_data) {
    AppData* data = (AppData*)user_data;

    switch (load_event) {
        case BLINK_LOAD_STARTED:
            gtk_label_set_text(GTK_LABEL(data->status_label),
                              "Loading...");
            break;

        case BLINK_LOAD_FINISHED:
            data->load_count++;
            char* message = g_strdup_printf("Done (#%d)", data->load_count);
            gtk_label_set_text(GTK_LABEL(data->status_label), message);
            g_free(message);
            break;

        default:
            break;
    }
}

static void on_activate(GtkApplication* app, gpointer user_data) {
    AppData* data = g_new0(AppData, 1);

    data->window = gtk_application_window_new(app);
    data->status_label = gtk_label_new("Ready");

    GtkWidget* web_view = blink_web_view_new();
    GtkWidget* vbox = gtk_box_new(GTK_ORIENTATION_VERTICAL, 0);

    gtk_box_append(GTK_BOX(vbox), data->status_label);
    gtk_box_append(GTK_BOX(vbox), web_view);

    /* Connect signal (pass data as user_data) */
    g_signal_connect(web_view, "load-changed",
                     G_CALLBACK(on_load_changed), data);

    gtk_window_set_child(GTK_WINDOW(data->window), vbox);
    gtk_window_set_default_size(GTK_WINDOW(data->window), 800, 600);

    blink_web_view_load_uri(BLINK_WEB_VIEW(web_view), "https://example.com");
    gtk_window_present(GTK_WINDOW(data->window));
}

int main(int argc, char* argv[]) {
    blink_gtk_init(&argc, &argv);  /* GTK4 は引数なし */

    GtkApplication* app = gtk_application_new("org.example.userdata", 0);
    g_signal_connect(app, "activate", G_CALLBACK(on_activate), NULL);

    /* ⚠️ g_application_run() は使用禁止 — Chromium RunLoop が飢餓状態になる
     * g_application_register/activate で GtkApplication シグナルを発火させてから
     * blink_gtk_run_main_loop() でChromium RunLoop を起動する (Build 658実証済み) */
    g_application_register(G_APPLICATION(app), NULL, NULL);
    g_application_activate(G_APPLICATION(app));
    int status = blink_gtk_run_main_loop();

    g_object_unref(app);
    blink_gtk_shutdown();
    return status;
}

Key Points:

  • Manage application state with AppData struct
  • Pass data as 4th argument to g_signal_connect()
  • Cast user_data inside handler

Beginner - Using Single Signals

Example 1: Progress Bar Display

Update progress bar with load-changed signal:

#include <gtk/gtk.h>
#include <blink_gtk/blink_gtk.h>

typedef struct {
    GtkWidget* progress_bar;
    GtkWidget* status_label;
} ProgressData;

static void on_load_changed(GtkWidget* web_view,
                            BlinkLoadEvent load_event,
                            gpointer user_data) {
    ProgressData* data = (ProgressData*)user_data;

    switch (load_event) {
        case BLINK_LOAD_STARTED:
            /* 0% */
            gtk_progress_bar_set_fraction(
                GTK_PROGRESS_BAR(data->progress_bar), 0.0);
            gtk_widget_set_visible(data->progress_bar, TRUE);
            gtk_label_set_text(GTK_LABEL(data->status_label),
                              "Loading started...");
            break;

        case BLINK_LOAD_REDIRECTED:
            /* 30% */
            gtk_progress_bar_set_fraction(
                GTK_PROGRESS_BAR(data->progress_bar), 0.3);
            gtk_label_set_text(GTK_LABEL(data->status_label),
                              "Redirecting...");
            break;

        case BLINK_LOAD_COMMITTED:
            /* 60% */
            gtk_progress_bar_set_fraction(
                GTK_PROGRESS_BAR(data->progress_bar), 0.6);
            gtk_label_set_text(GTK_LABEL(data->status_label),
                              "Rendering...");
            break;

        case BLINK_LOAD_FINISHED:
            /* 100% */
            gtk_progress_bar_set_fraction(
                GTK_PROGRESS_BAR(data->progress_bar), 1.0);
            gtk_label_set_text(GTK_LABEL(data->status_label), "Done");

            /* Hide progress bar after 1 second */
            g_timeout_add(1000, (GSourceFunc)gtk_widget_hide,
                         data->progress_bar);
            break;
    }
}

static void on_activate(GtkApplication* app, gpointer user_data) {
    ProgressData* data = g_new0(ProgressData, 1);

    GtkWidget* window = gtk_application_window_new(app);
    GtkWidget* web_view = blink_web_view_new();

    data->progress_bar = gtk_progress_bar_new();
    data->status_label = gtk_label_new("Ready");

    gtk_widget_set_visible(data->progress_bar, FALSE);

    GtkWidget* vbox = gtk_box_new(GTK_ORIENTATION_VERTICAL, 0);
    gtk_box_append(GTK_BOX(vbox), data->progress_bar);
    gtk_box_append(GTK_BOX(vbox), data->status_label);
    gtk_box_append(GTK_BOX(vbox), web_view);

    /* Connect signal */
    g_signal_connect(web_view, "load-changed",
                     G_CALLBACK(on_load_changed), data);

    gtk_window_set_child(GTK_WINDOW(window), vbox);
    gtk_window_set_default_size(GTK_WINDOW(window), 800, 600);

    blink_web_view_load_uri(BLINK_WEB_VIEW(web_view), "https://www.chromium.org");
    gtk_window_present(GTK_WINDOW(window));
}

int main(int argc, char* argv[]) {
    blink_gtk_init(&argc, &argv);  /* GTK4 は引数なし */

    GtkApplication* app = gtk_application_new("org.example.progress", 0);
    g_signal_connect(app, "activate", G_CALLBACK(on_activate), NULL);

    /* ⚠️ g_application_run() は使用禁止 — Chromium RunLoop が飢餓状態になる
     * g_application_register/activate で GtkApplication シグナルを発火させてから
     * blink_gtk_run_main_loop() でChromium RunLoop を起動する (Build 658実証済み) */
    g_application_register(G_APPLICATION(app), NULL, NULL);
    g_application_activate(G_APPLICATION(app));
    int status = blink_gtk_run_main_loop();

    g_object_unref(app);
    blink_gtk_shutdown();
    return status;
}

Behavior:

  1. Load start: Show progress bar, 0%
  2. Redirect: 30%
  3. Commit: 60%
  4. Finish: 100%, hide after 1 second

Example 2: Error Handling

Display error dialog with load-failed signal:

#include <gtk/gtk.h>
#include <blink_gtk/blink_gtk.h>

static void on_load_failed(GtkWidget* web_view,
                          const char* failing_url,
                          int error_code,
                          gpointer user_data) {
    GtkWindow* window = GTK_WINDOW(user_data);

    /* Generate error message */
    const char* error_message = "Unknown error";

    switch (error_code) {
        case -2:
            error_message = "Failed to load page";
            break;
        case -7:
            error_message = "Connection timed out";
            break;
        case -105:
            error_message = "Server not found";
            break;
        case -106:
            error_message = "No internet connection";
            break;
    }

    /* Show error dialog */
    GtkWidget* dialog = gtk_message_dialog_new(
        window,
        GTK_DIALOG_MODAL,
        GTK_MESSAGE_ERROR,
        GTK_BUTTONS_OK,
        "Loading Error"
    );

    gtk_message_dialog_format_secondary_text(
        GTK_MESSAGE_DIALOG(dialog),
        "%s\n\nURL: %s\nError code: %d",
        error_message, failing_url, error_code
    );

    g_signal_connect(dialog, "response",
                     G_CALLBACK(gtk_window_destroy), NULL);

    gtk_widget_show(dialog);
}

static void on_activate(GtkApplication* app, gpointer user_data) {
    GtkWidget* window = gtk_application_window_new(app);
    GtkWidget* web_view = blink_web_view_new();

    /* Connect signal */
    g_signal_connect(web_view, "load-failed",
                     G_CALLBACK(on_load_failed), window);

    gtk_window_set_child(GTK_WINDOW(window), web_view);
    gtk_window_set_default_size(GTK_WINDOW(window), 800, 600);

    /* Test with non-existent URL */
    blink_web_view_load_uri(BLINK_WEB_VIEW(web_view), "https://this-does-not-exist.example");
    gtk_window_present(GTK_WINDOW(window));
}

int main(int argc, char* argv[]) {

    GtkApplication* app = gtk_application_new("org.example.error", 0);

     * g_application_register/activate で GtkApplication シグナルを発火させてから
     * blink_gtk_run_main_loop() でChromium RunLoop を起動する (Build 658実証済み) */

    g_application_activate(G_APPLICATION(app));

Intermediate - Combining Multiple Signals

Building Browser UI

Practical browser UI combining multiple signals:

#include <gtk/gtk.h>
#include <blink_gtk/blink_gtk.h>

/* Application data */
typedef struct {
    GtkWidget* window;
    GtkWidget* url_entry;
    GtkWidget* back_button;
    GtkWidget* forward_button;
    GtkWidget* reload_button;
    GtkWidget* stop_button;
    GtkWidget* progress_bar;
    GtkWidget* status_label;
    GtkWidget* web_view;
} BrowserUI;

/* load-changed: Update UI based on load state */
static void on_load_changed(GtkWidget* web_view,
                            BlinkLoadEvent load_event,
                            gpointer user_data) {
    BrowserUI* ui = (BrowserUI*)user_data;

    switch (load_event) {
        case BLINK_LOAD_STARTED:
            /* Show progress bar */
            gtk_progress_bar_set_fraction(
                GTK_PROGRESS_BAR(ui->progress_bar), 0.0);
            gtk_widget_set_visible(ui->progress_bar, TRUE);

            /* Show stop button, hide reload button */
            gtk_widget_set_visible(ui->stop_button, TRUE);
            gtk_widget_set_visible(ui->reload_button, FALSE);

            gtk_label_set_text(GTK_LABEL(ui->status_label), "Loading...");
            break;

        case BLINK_LOAD_COMMITTED:
            gtk_progress_bar_set_fraction(
                GTK_PROGRESS_BAR(ui->progress_bar), 0.5);
            break;

        case BLINK_LOAD_FINISHED: {
            gtk_progress_bar_set_fraction(
                GTK_PROGRESS_BAR(ui->progress_bar), 1.0);
            gtk_widget_set_visible(ui->progress_bar, FALSE);

            /* Show reload button, hide stop button */
            gtk_widget_set_visible(ui->stop_button, FALSE);
            gtk_widget_set_visible(ui->reload_button, TRUE);

            gtk_label_set_text(GTK_LABEL(ui->status_label), "Done");

            /* Update navigation button states */
            gboolean can_go_back = blink_web_view_can_go_back(BLINK_WEB_VIEW(web_view));
            gboolean can_go_forward = blink_web_view_can_go_forward(BLINK_WEB_VIEW(web_view));

            gtk_widget_set_sensitive(ui->back_button, can_go_back);
            gtk_widget_set_sensitive(ui->forward_button, can_go_forward);
            break;
        }

        default:
            break;
    }
}

/* load-failed: Show error */
static void on_load_failed(GtkWidget* web_view,
                          const char* failing_url,
                          int error_code,
                          gpointer user_data) {
    BrowserUI* ui = (BrowserUI*)user_data;

    char* message = g_strdup_printf("Error (code: %d)", error_code);
    gtk_label_set_text(GTK_LABEL(ui->status_label), message);
    g_free(message);

    gtk_widget_set_visible(ui->progress_bar, FALSE);
    gtk_widget_set_visible(ui->stop_button, FALSE);
    gtk_widget_set_visible(ui->reload_button, TRUE);
}

/* title-changed: Update window title */
static void on_title_changed(GtkWidget* web_view,
                            const char* title,
                            gpointer user_data) {
    BrowserUI* ui = (BrowserUI*)user_data;

    char* window_title = g_strdup_printf("%s - MyBrowser", title);
    gtk_window_set_title(GTK_WINDOW(ui->window), window_title);
    g_free(window_title);
}

/* uri-changed: Update address bar */
static void on_uri_changed(GtkWidget* web_view,
                          const char* uri,
                          gpointer user_data) {
    BrowserUI* ui = (BrowserUI*)user_data;

    gtk_editable_set_text(GTK_EDITABLE(ui->url_entry), uri);
}

/* Button click handlers */
static void on_back_clicked(GtkButton* button, gpointer user_data) {
    BrowserUI* ui = (BrowserUI*)user_data;
    blink_web_view_go_back(BLINK_WEB_VIEW(ui->web_view));
}

static void on_forward_clicked(GtkButton* button, gpointer user_data) {
    BrowserUI* ui = (BrowserUI*)user_data;
    blink_web_view_go_forward(BLINK_WEB_VIEW(ui->web_view));
}

static void on_reload_clicked(GtkButton* button, gpointer user_data) {
    BrowserUI* ui = (BrowserUI*)user_data;
    blink_web_view_reload(BLINK_WEB_VIEW(ui->web_view));
}

static void on_stop_clicked(GtkButton* button, gpointer user_data) {
    BrowserUI* ui = (BrowserUI*)user_data;
    blink_web_view_stop(BLINK_WEB_VIEW(ui->web_view));
}

static void on_url_activate(GtkEntry* entry, gpointer user_data) {
    BrowserUI* ui = (BrowserUI*)user_data;
    const char* url = gtk_editable_get_text(GTK_EDITABLE(entry));
    blink_web_view_load_uri(BLINK_WEB_VIEW(ui->web_view), url);
}

/* Connect signals */
static void setup_signals(BrowserUI* ui) {
    /* WebView signals */
    g_signal_connect(ui->web_view, "load-changed",
                     G_CALLBACK(on_load_changed), ui);
    g_signal_connect(ui->web_view, "load-failed",
                     G_CALLBACK(on_load_failed), ui);
    g_signal_connect(ui->web_view, "title-changed",
                     G_CALLBACK(on_title_changed), ui);
    g_signal_connect(ui->web_view, "uri-changed",
                     G_CALLBACK(on_uri_changed), ui);

    /* Button signals */
    g_signal_connect(ui->back_button, "clicked",
                     G_CALLBACK(on_back_clicked), ui);
    g_signal_connect(ui->forward_button, "clicked",
                     G_CALLBACK(on_forward_clicked), ui);
    g_signal_connect(ui->reload_button, "clicked",
                     G_CALLBACK(on_reload_clicked), ui);
    g_signal_connect(ui->stop_button, "clicked",
                     G_CALLBACK(on_stop_clicked), ui);
    g_signal_connect(ui->url_entry, "activate",
                     G_CALLBACK(on_url_activate), ui);
}

static void on_activate(GtkApplication* app, gpointer user_data) {
    BrowserUI* ui = g_new0(BrowserUI, 1);

    /* Window */
    ui->window = gtk_application_window_new(app);
    gtk_window_set_title(GTK_WINDOW(ui->window), "MyBrowser");
    gtk_window_set_default_size(GTK_WINDOW(ui->window), 1024, 768);

    /* Toolbar */
    GtkWidget* toolbar = gtk_box_new(GTK_ORIENTATION_HORIZONTAL, 5);
    gtk_widget_set_margin_start(toolbar, 5);
    gtk_widget_set_margin_end(toolbar, 5);
    gtk_widget_set_margin_top(toolbar, 5);
    gtk_widget_set_margin_bottom(toolbar, 5);

    ui->back_button = gtk_button_new_from_icon_name("go-previous");
    ui->forward_button = gtk_button_new_from_icon_name("go-next");
    ui->reload_button = gtk_button_new_from_icon_name("view-refresh");
    ui->stop_button = gtk_button_new_from_icon_name("process-stop");
    ui->url_entry = gtk_entry_new();

    gtk_widget_set_visible(ui->stop_button, FALSE);
    gtk_widget_set_sensitive(ui->back_button, FALSE);
    gtk_widget_set_sensitive(ui->forward_button, FALSE);

    gtk_widget_set_hexpand(ui->url_entry, TRUE);

    gtk_box_append(GTK_BOX(toolbar), ui->back_button);
    gtk_box_append(GTK_BOX(toolbar), ui->forward_button);
    gtk_box_append(GTK_BOX(toolbar), ui->reload_button);
    gtk_box_append(GTK_BOX(toolbar), ui->stop_button);
    gtk_box_append(GTK_BOX(toolbar), ui->url_entry);

    /* Progress bar */
    ui->progress_bar = gtk_progress_bar_new();
    gtk_widget_set_visible(ui->progress_bar, FALSE);

    /* Status bar */
    ui->status_label = gtk_label_new("Ready");
    gtk_widget_set_halign(ui->status_label, GTK_ALIGN_START);
    gtk_widget_set_margin_start(ui->status_label, 5);
    gtk_widget_set_margin_bottom(ui->status_label, 2);

    /* WebView */
    ui->web_view = blink_web_view_new();

    /* Layout */
    GtkWidget* vbox = gtk_box_new(GTK_ORIENTATION_VERTICAL, 0);
    gtk_box_append(GTK_BOX(vbox), toolbar);
    gtk_box_append(GTK_BOX(vbox), ui->progress_bar);
    gtk_box_append(GTK_BOX(vbox), ui->web_view);
    gtk_box_append(GTK_BOX(vbox), ui->status_label);

    gtk_widget_set_vexpand(ui->web_view, TRUE);

    gtk_window_set_child(GTK_WINDOW(ui->window), vbox);

    /* Connect signals */
    setup_signals(ui);

    /* Initial URL */
    blink_web_view_load_uri(BLINK_WEB_VIEW(ui->web_view), "https://www.chromium.org");

    gtk_window_present(GTK_WINDOW(ui->window));
}

int main(int argc, char* argv[]) {
    blink_gtk_init(&argc, &argv);  /* GTK4 は引数なし */

    GtkWidget* back_button;
    GtkWidget* forward_button;
    GtkWidget* reload_button;
    GtkWidget* stop_button;
    GtkWidget* progress_bar;
    GtkWidget* status_label;
    GtkWidget* security_icon;

    /* Settings */
    GtkWidget* js_switch;
    GtkWidget* images_switch;
    GtkWidget* storage_switch;
} AdvancedBrowserUI;

/* load-changed */
static void on_load_changed(GtkWidget* web_view,
                            BlinkLoadEvent load_event,
                            gpointer user_data) {
    AdvancedBrowserUI* ui = (AdvancedBrowserUI*)user_data;

    switch (load_event) {
        case BLINK_LOAD_STARTED:
            gtk_progress_bar_set_fraction(GTK_PROGRESS_BAR(ui->progress_bar), 0.0);
            gtk_widget_set_visible(ui->progress_bar, TRUE);
            gtk_widget_set_visible(ui->stop_button, TRUE);
            gtk_widget_set_visible(ui->reload_button, FALSE);
            gtk_label_set_text(GTK_LABEL(ui->status_label), "Loading...");
            break;

        case BLINK_LOAD_COMMITTED:
            gtk_progress_bar_set_fraction(GTK_PROGRESS_BAR(ui->progress_bar), 0.5);
            break;

        case BLINK_LOAD_FINISHED:
            gtk_progress_bar_set_fraction(GTK_PROGRESS_BAR(ui->progress_bar), 1.0);
            gtk_widget_set_visible(ui->progress_bar, FALSE);
            gtk_widget_set_visible(ui->stop_button, FALSE);
            gtk_widget_set_visible(ui->reload_button, TRUE);
            gtk_label_set_text(GTK_LABEL(ui->status_label), "Done");

            gboolean can_back = blink_web_view_can_go_back(BLINK_WEB_VIEW(web_view));
            gboolean can_forward = blink_web_view_can_go_forward(BLINK_WEB_VIEW(web_view));
            gtk_widget_set_sensitive(ui->back_button, can_back);
            gtk_widget_set_sensitive(ui->forward_button, can_forward);
            break;

        default:
            break;
    }
}

/* load-failed */
static void on_load_failed(GtkWidget* web_view,
                          const char* failing_url,
                          int error_code,
                          gpointer user_data) {
    AdvancedBrowserUI* ui = (AdvancedBrowserUI*)user_data;

    const char* error_msg = "Unknown error";
    switch (error_code) {
        case -2: error_msg = "Loading failed"; break;
        case -7: error_msg = "Timed out"; break;
        case -105: error_msg = "Server not found"; break;
        case -106: error_msg = "Internet disconnected"; break;
    }

    char* message = g_strdup_printf("%s (code: %d)", error_msg, error_code);
    gtk_label_set_text(GTK_LABEL(ui->status_label), message);
    g_free(message);

    /* Error dialog */
    GtkWidget* dialog = gtk_message_dialog_new(
        GTK_WINDOW(ui->window),
        GTK_DIALOG_MODAL,
        GTK_MESSAGE_ERROR,
        GTK_BUTTONS_OK,
        "Loading Error"
    );
    gtk_message_dialog_format_secondary_text(
        GTK_MESSAGE_DIALOG(dialog),
        "%s\n\nURL: %s\nError code: %d",
        error_msg, failing_url, error_code
    );
    g_signal_connect(dialog, "response", G_CALLBACK(gtk_window_destroy), NULL);
    gtk_widget_show(dialog);

    gtk_widget_set_visible(ui->progress_bar, FALSE);
    gtk_widget_set_visible(ui->stop_button, FALSE);
    gtk_widget_set_visible(ui->reload_button, TRUE);
}

/* title-changed */
static void on_title_changed(GtkWidget* web_view,
                            const char* title,
                            gpointer user_data) {

/* uri-changed */
static void on_uri_changed(GtkWidget* web_view,
                          const char* uri,
                          gpointer user_data) {
    AdvancedBrowserUI* ui = (AdvancedBrowserUI*)user_data;

    /* Update address bar */
    gtk_editable_set_text(GTK_EDITABLE(ui->url_entry), uri);

    /* Update security icon */
    if (g_str_has_prefix(uri, "https://")) {
        gtk_image_set_from_icon_name(GTK_IMAGE(ui->security_icon),
                                     "dialog-password");
        gtk_widget_set_tooltip_text(ui->security_icon,
                                    "This connection is secure");
    } else {
        gtk_image_set_from_icon_name(GTK_IMAGE(ui->security_icon),
                                     "dialog-warning");
        gtk_widget_set_tooltip_text(ui->security_icon,
                                    "This connection is not secure");
    }
}

/* Signal connection */
static void setup_signals(AdvancedBrowserUI* ui) {
    /* WebView signals */
    g_signal_connect(ui->web_view, "load-changed",
                     G_CALLBACK(on_load_changed), ui);
    g_signal_connect(ui->web_view, "load-failed",
                     G_CALLBACK(on_load_failed), ui);
    g_signal_connect(ui->web_view, "title-changed",
                     G_CALLBACK(on_title_changed), ui);
    g_signal_connect(ui->web_view, "uri-changed",
                     G_CALLBACK(on_uri_changed), ui);

    /* Buttons */
    g_signal_connect(ui->back_button, "clicked",
                     G_CALLBACK(on_back_clicked), ui);
    g_signal_connect(ui->forward_button, "clicked",
                     G_CALLBACK(on_forward_clicked), ui);
    g_signal_connect(ui->reload_button, "clicked",
                     G_CALLBACK(on_reload_clicked), ui);
    g_signal_connect(ui->stop_button, "clicked",
                     G_CALLBACK(on_stop_clicked), ui);
    g_signal_connect(ui->url_entry, "activate",
                     G_CALLBACK(on_url_activate), ui);

    /* Settings */
    g_signal_connect(ui->js_switch, "notify::active",
                     G_CALLBACK(on_js_switch_changed), ui);
    g_signal_connect(ui->images_switch, "notify::active",
                     G_CALLBACK(on_images_switch_changed), ui);
    g_signal_connect(ui->storage_switch, "notify::active",
                     G_CALLBACK(on_storage_switch_changed), ui);
}

    gtk_box_append(GTK_BOX(settings_box), gtk_label_new("JavaScript:"));

    setup_signals(ui);

    /* Initial URL */
    blink_web_view_load_uri(BLINK_WEB_VIEW(ui->web_view), "https://www.chromium.org");

    gtk_window_present(GTK_WINDOW(ui->window));
}

int main(int argc, char* argv[]) {
    blink_gtk_init(&argc, &argv);  /* GTK4 は引数なし */

    GtkApplication* app = gtk_application_new("org.example.advanced", 0);
    g_signal_connect(app, "activate", G_CALLBACK(on_activate), NULL);

    /* ⚠️ g_application_run() は使用禁止 — Chromium RunLoop が飢餓状態になる
     * g_application_register/activate で GtkApplication シグナルを発火させてから
     * blink_gtk_run_main_loop() でChromium RunLoop を起動する (Build 658実証済み) */
    g_application_register(G_APPLICATION(app), NULL, NULL);
    g_application_activate(G_APPLICATION(app));
    int status = blink_gtk_run_main_loop();

    g_object_unref(app);
    blink_gtk_shutdown();
    return status;
}

Pattern Collection

Pattern 1: Loading Indicator

Simple spinner display:

typedef struct {
    GtkWidget* spinner;
    GtkWidget* web_view;
} SpinnerData;

static void on_load_changed(GtkWidget* web_view,
                            BlinkLoadEvent load_event,
                            gpointer user_data) {
    SpinnerData* data = (SpinnerData*)user_data;

    if (load_event == BLINK_LOAD_STARTED) {
        gtk_spinner_start(GTK_SPINNER(data->spinner));
        gtk_widget_set_visible(data->spinner, TRUE);
    } else if (load_event == BLINK_LOAD_FINISHED) {
        gtk_spinner_stop(GTK_SPINNER(data->spinner));
        gtk_widget_set_visible(data->spinner, FALSE);
    }
}

Pattern 2: History Management

Simple history list:

typedef struct {
    GList* history;  // String list
    int current_index;
} HistoryManager;

static void on_uri_changed(GtkWidget* web_view,
                          const char* uri,
                          gpointer user_data) {
    HistoryManager* history = (HistoryManager*)user_data;

    /* Remove forward history */
    while (history->current_index < g_list_length(history->history) - 1) {
        GList* last = g_list_last(history->history);
        g_free(last->data);
        history->history = g_list_delete_link(history->history, last);
    }

    /* Add new URI */
    history->history = g_list_append(history->history, g_strdup(uri));
    history->current_index = g_list_length(history->history) - 1;
}

Pattern 3: Title and Favicon

Title and simple favicon display:

typedef struct {
    GtkWidget* tab_label;
    GtkWidget* tab_icon;
} TabData;

static void on_title_changed(GtkWidget* web_view,
                            const char* title,
                            gpointer user_data) {
    TabData* tab = (TabData*)user_data;

    /* Truncate long titles */
    char* display_title;
    if (strlen(title) > 20) {
        display_title = g_strdup_printf("%.17s...", title);
    } else {
        display_title = g_strdup(title);
    }

    gtk_label_set_text(GTK_LABEL(tab->tab_label), display_title);
    g_free(display_title);
}

static void on_uri_changed(GtkWidget* web_view,
                          const char* uri,
                          gpointer user_data) {
    TabData* tab = (TabData*)user_data;

    /* Lock icon for HTTPS pages */
    if (g_str_has_prefix(uri, "https://")) {
        gtk_image_set_from_icon_name(GTK_IMAGE(tab->tab_icon),
                                     "dialog-password");
    } else {
        gtk_image_set_from_icon_name(GTK_IMAGE(tab->tab_icon),
                                     "text-html");
    }
}

Troubleshooting

Signals Not Firing

Symptom: Signal handler not being called

Causes and Solutions:

  1. Signal name typo

    // ❌ Wrong
g_signal_connect(web_view, "load_changed", ...);  // Underscore

// ✅ Correct
g_signal_connect(web_view, "load-changed", ...);  // Hyphen

  2. Connecting before WebView initialization

    // ❌ Wrong
g_signal_connect(web_view, "load-changed", ...);
web_view = blink_web_view_new();

// ✅ Correct
web_view = blink_web_view_new();
g_signal_connect(web_view, "load-changed", ...);

  3. GTK main loop not running

    blink_web_view_load_uri(BLINK_WEB_VIEW(web_view), "https://example.com");
// ❌ Signals won't fire (no main loop)

blink_gtk_run_main_loop();  // ✅ Start main loop (g_application_run() is forbidden)

user_data is NULL (Crashes)

Symptom: Segmentation fault in signal handler

Solution:

static void on_load_changed(GtkWidget* web_view,
                            BlinkLoadEvent load_event,
                            gpointer user_data) {
    /* NULL check */
    if (user_data == NULL) {
        g_warning("user_data is NULL");
        return;
    }

    MyData* data = (MyData*)user_data;
    /* ... */
}

Memory Leaks

Symptom: Memory usage keeps growing

Causes and Solutions:

  1. String leaks

    /* ❌ Memory leak */
const char* ua = blink_web_view_get_user_agent(web_view);

/* ✅ Correct */
char* ua = blink_web_view_get_user_agent(web_view);
g_print("UA: %s\n", ua);
g_free(ua);  // Required

  2. Signal handler not disconnected

    /* Save handler ID */
gulong handler_id = g_signal_connect(web_view, "load-changed",
                                     G_CALLBACK(on_load_changed), NULL);

/* Disconnect before destroying WebView */
g_signal_handler_disconnect(web_view, handler_id);

Progress Bar Not Updating

Symptom: Progress bar stays at 0%

Solution:

static void on_load_changed(GtkWidget* web_view,
                            BlinkLoadEvent load_event,
                            gpointer user_data) {
    ProgressData* data = (ProgressData*)user_data;

    /* Handle all events */
    switch (load_event) {
        case BLINK_LOAD_STARTED:
            gtk_progress_bar_set_fraction(
                GTK_PROGRESS_BAR(data->progress_bar), 0.0);
            gtk_widget_set_visible(data->progress_bar, TRUE);  // Don't forget to show
            break;

        case BLINK_LOAD_COMMITTED:
            gtk_progress_bar_set_fraction(
                GTK_PROGRESS_BAR(data->progress_bar), 0.5);
            break;

        case BLINK_LOAD_FINISHED:
            gtk_progress_bar_set_fraction(
                GTK_PROGRESS_BAR(data->progress_bar), 1.0);
            break;
    }
}
