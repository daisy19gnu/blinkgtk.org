/* セッションファイルのパスを返す（呼び出し元でg_freeする） */
static char* get_session_path(void) {
  const char* config_dir = g_get_user_config_dir();
  char dir_path[4096];
  snprintf(dir_path, sizeof(dir_path), "%s/%s", config_dir, SESSION_INI_SUBDIR);
  g_mkdir_with_parents(dir_path, 0700);
  char* file_path = g_strdup_printf("%s/%s/%s", config_dir, SESSION_INI_SUBDIR, SESSION_INI_FILE);
  return file_path;
}

/* セッション情報をINIファイルに保存する
 * execv()再起動前に呼ぶことでシームレス復元を実現（CP-843） */
static void save_session(void) {
  GtkWindow* win = NULL;
  if (g_web_view) {
    GtkRoot* root = gtk_widget_get_root(GTK_WIDGET(g_web_view));
    if (root && GTK_IS_WINDOW(root)) {
      win = GTK_WINDOW(root);
    }
  }

  int w = 800, h = 600;
  if (win) {
    gtk_window_get_default_size(win, &w, &h);
    /* get_default_sizeが-1を返す場合は実サイズを取得 */
    if (w <= 0 || h <= 0) {
      /* GTK4ではwidthとheightを直接取得 */
      w = gtk_widget_get_width(GTK_WIDGET(win));
      h = gtk_widget_get_height(GTK_WIDGET(win));
      if (w <= 0) w = 800;
      if (h <= 0) h = 600;
    }
  }

  char* uri = NULL;
  if (g_web_view) {
    g_object_get(g_web_view, "uri", &uri, NULL);
  }

  const char* gpu_mode = getenv("BLINKGTK_GPU_MODE");
  gboolean ebook_active = g_ebook_check ?
      gtk_check_button_get_active(g_ebook_check) : FALSE;

  char* path = get_session_path();
  FILE* fp = fopen(path, "w");
  if (fp) {
    fprintf(fp, "[session]\n");
    fprintf(fp, "window_width=%d\n",  w > 0 ? w : 800);
    fprintf(fp, "window_height=%d\n", h > 0 ? h : 600);
    fprintf(fp, "url=%s\n",      uri      ? uri      : "https://www.google.com");
    fprintf(fp, "gpu_mode=%s\n", gpu_mode ? gpu_mode : "software");
    fprintf(fp, "ebook_active=%d\n", ebook_active ? 1 : 0);
    fprintf(fp, "scroll_x=%d\n", g_cached_scroll_x);
    fprintf(fp, "scroll_y=%d\n", g_cached_scroll_y);
    fclose(fp);
    printf("[CP-842] Session saved: %dx%d url=%s gpu=%s ebook=%d scroll=%d,%d\n",
           w, h, uri ? uri : "(none)", gpu_mode ? gpu_mode : "software",
           ebook_active ? 1 : 0, g_cached_scroll_x, g_cached_scroll_y);
  } else {
    printf("[CP-842] WARNING: Could not save session to %s\n", path);
  }
  g_free(path);
  g_free(uri);
}

/* セッションINIファイルを読み込み、各値をポインタに格納する
 * 読み込み成功なら1、失敗なら0を返す。
 * out_url は呼び出し元でfree()すること。 */
static int load_session(int* out_w, int* out_h, char** out_url,
                        char** out_gpu, int* out_ebook,
                        int* out_sx, int* out_sy) {
  char* path = get_session_path();
  FILE* fp = fopen(path, "r");
  g_free(path);
  if (!fp) return 0;

  /* デフォルト値 */
  *out_w     = 800;
  *out_h     = 600;
  *out_url   = NULL;
  *out_gpu   = NULL;
  *out_ebook = 0;
  *out_sx    = 0;
  *out_sy    = 0;

  char line[4096];
  while (fgets(line, sizeof(line), fp)) {
    /* 末尾の改行を除去 */
    size_t len = strlen(line);
    while (len > 0 && (line[len-1] == '\n' || line[len-1] == '\r')) {
      line[--len] = '\0';
    }
    if (line[0] == '[') continue; /* セクションヘッダ */

    char key[256], value[4096];
    /* %4095[^\n] でスペースを含む値（data: URL等）を正しく読み込む */
    if (sscanf(line, "%255[^=]=%4095[^\n]", key, value) != 2) continue;

    if (strcmp(key, "window_width")  == 0) *out_w     = atoi(value);
    else if (strcmp(key, "window_height") == 0) *out_h = atoi(value);
    else if (strcmp(key, "url")      == 0) *out_url   = strdup(value);
    else if (strcmp(key, "gpu_mode") == 0) {
      // 5/11 Lead 指摘: BlinkGTK 正規 GPU mode は egl。旧 "swiftshader" 設定値が
      // session.ini に残っている場合、"egl" に自動 migration (レビュアー C 指摘)。
      if (strcmp(value, "swiftshader") == 0) {
        *out_gpu = strdup("egl");
        printf("[CP-844] session.ini gpu=swiftshader → egl migration (BlinkGTK 正規 mode)\n");
      } else {
        *out_gpu = strdup(value);
      }
    }
    else if (strcmp(key, "ebook_active") == 0) *out_ebook = atoi(value);
    else if (strcmp(key, "scroll_x") == 0) *out_sx    = atoi(value);
    else if (strcmp(key, "scroll_y") == 0) *out_sy    = atoi(value);
  }
  fclose(fp);

  /* 最低限の検証 */
  if (*out_w < 200 || *out_w > 7680) *out_w = 800;
  if (*out_h < 100 || *out_h > 4320) *out_h = 600;

  printf("[CP-844] Session loaded: %dx%d url=%s gpu=%s ebook=%d scroll=%d,%d\n",
         *out_w, *out_h, *out_url ? *out_url : "(none)",
         *out_gpu ? *out_gpu : "(none)", *out_ebook, *out_sx, *out_sy);
  return 1;
}

/* ======== CP-846: スクロール位置キャッシュタイマー ======== */

/* JavaScript経由でスクロール位置を取得するAPI（dlsymで動的検索） */
typedef void (*EvalJsFn)(BlinkWebView*, const char*, void (*)(const char*, void*), void*);
static EvalJsFn g_eval_js_fn = NULL;
static int g_eval_js_tried = 0;

static EvalJsFn get_eval_js(void) {
  if (!g_eval_js_tried) {
    g_eval_js_tried = 1;
    /* blink_web_view_execute_javascript(view, script, callback, data) */
    g_eval_js_fn = (EvalJsFn)dlsym(NULL, "blink_web_view_execute_javascript");
  }
  return g_eval_js_fn;
}

/* JS実行コールバック: "scrollX,scrollY" 形式を期待 */
static void on_scroll_result(const char* result, void* user_data) {
  (void)user_data;
  if (!result) return;
  int sx = 0, sy = 0;
  if (sscanf(result, "%d,%d", &sx, &sy) == 2) {
    g_cached_scroll_x = sx;
    g_cached_scroll_y = sy;
  }
}

/* 500ms周期のスクロール位置キャッシュタイマー */
static gboolean on_scroll_cache_timer(gpointer user_data) {
  (void)user_data;
  if (!g_web_view) return G_SOURCE_CONTINUE;

  EvalJsFn eval_js = get_eval_js();
  if (eval_js) {
    /* "X,Y" 形式で返す簡易スクリプト */
    eval_js(g_web_view,
            "window.scrollX + ',' + window.scrollY",
            on_scroll_result, NULL);
  }
  return G_SOURCE_CONTINUE;
}

/* ======== CP-845: ページロード完了後のスクロール復元 ======== */
/* CP-849修正: notify::uriはload_uri()内でURIセット直後（LoadURL前）に発火するため
 * window.scrollTo()を呼んでも早すぎて効果がなかった。load-changedシグナルの
 * BLINK_LOAD_FINISHED (3) を使い、ページDOM確定後にスクロールを復元する。 */

/* CP-849: load-changed シグナルの正しいシグネチャ (GtkWidget*, gint, gpointer) */
static void on_load_changed_for_scroll(GtkWidget* widget,
                                        gint load_event,
                                        gpointer user_data) {
  (void)user_data;
  /* BLINK_LOAD_FINISHED = 2: DOM確定後のタイミングでスクロール復元
   * (実際にemitされる値は2。blink_web_view.cc:558 g_signal_emit_by_name(..., 2) 参照) */
  if (load_event != 2) return;

  BlinkWebView* view = BLINK_WEB_VIEW(widget);

  /* スクロール復元 */
  if (g_restore_session &&
      (g_restore_scroll_x != 0 || g_restore_scroll_y != 0)) {
    EvalJsFn eval_js = get_eval_js();
    if (eval_js && view) {
      char script[256];
      snprintf(script, sizeof(script),
               "window.scrollTo(%d, %d);", g_restore_scroll_x, g_restore_scroll_y);
      eval_js(view, script, NULL, NULL);
      printf("[CP-845] Scroll restored: (%d, %d)\n",
             g_restore_scroll_x, g_restore_scroll_y);
      g_restore_scroll_x = 0;
      g_restore_scroll_y = 0;
    }
  }

  /* Lead 指示 2026-05-04: ページロード完了 (DOM 確定) 時に JLReq ボタンが
   * ON ならツールチップ JS を再注入する。
   *
   * 理由: g_signal_connect("toggled", ...) が gtk_check_button_set_active(TRUE)
   * の後に行われるため、起動時の初期 ON では toggled signal が空打ちで終わる。
   * load-changed の DOM 確定タイミングで明示的に注入することで、
   *   1. 初回ロード時の自動展開
   *   2. ナビゲーション後の再展開 (URL 切替時の自動再適用)
   * の両方を保証する (idempotent: 注入時に内部で remove → apply)。
   * JLREQ_TOOLTIP_JS_APPLY と on_jlreq_js_result は前方宣言を file 冒頭で実施。 */
  if (g_jlreq_check && gtk_check_button_get_active(g_jlreq_check) && view) {
    blink_web_view_execute_javascript(view,
                                      JLREQ_TOOLTIP_JS_APPLY,
                                      on_jlreq_js_result,
                                      NULL);
    printf("[demo] JLReq JS injected on load-changed (FINISHED)\n");
  }
  /* EBOOK ボタンの状態に応じて user-select block CSS を再注入。
   * page navigation で <style> が破棄される可能性があるため毎回注入。
   * 前方宣言不要 (file scope の static const char[] を直接参照、本関数は
   * EBOOK_SELECT_BLOCK_JS_APPLY の定義 (line 540 付近) より前に書かれて
   * いるが C は前方参照可能 (extern 宣言は file 冒頭で行う必要)。 */
  if (g_ebook_check && gtk_check_button_get_active(g_ebook_check) && view) {
    extern const char EBOOK_SELECT_BLOCK_JS_APPLY[];
    blink_web_view_execute_javascript(view,
                                      EBOOK_SELECT_BLOCK_JS_APPLY,
                                      NULL,
                                      NULL);
  }
}

/* ======== CP-793/CP-794 + CP-843: GPU チェックボックスのトグル ======== */

/* CP-793/CP-794: GPU チェックボックスのトグル時コールバック
 * CP-843: execv()前にセッションを保存し、--restore-session を渡す。
 * 再起動後は save_session() の内容で window サイズ・URL・EBOOK状態が復元される。
 * Chromium GPUプロセスはプロセスライフタイムに一度しか初期化されないため、
 * ランタイム切替にはプロセス再起動が必要（レビュアーA-D合意）。 */
static void on_gpu_toggled(GtkCheckButton* check, gpointer user_data) {
  (void)user_data;
  gboolean active = gtk_check_button_get_active(check);
  // 5/11 Lead 指摘修正: BlinkGTK の正規 GPU mode は egl (swiftshader は iter16h
  //   Phase 2-A3-β で実態 software に rewrite 済の誤った設定値)。
  //   BlinkGTK プロジェクトルール GPU mode 説明: software (default) / swiftshader (実態 software) /
  //   egl (native EGL, requires real GPU) と整合。
  const char* mode = active ? "egl" : "software";

  /* 環境変数をセット */
  setenv("BLINKGTK_GPU_MODE", mode, 1);
  printf("[CP-843] GPU mode → %s — saving session and restarting via execv()\n", mode);

  /* CP-843: execv()前にセッション状態を保存 */
  save_session();

  /* execv()で自プロセスを再起動（--restore-sessionフラグを付与） */
  char* new_argv[] = { g_argv0, (char*)"--restore-session", NULL };
  execv("/proc/self/exe", new_argv);

  /* execv()が失敗した場合のフォールバック */
  perror("[CP-843] execv() failed");
}

/* ======== CP-1063 描画経路セレクタ コールバック (Lead 6/3 採択) ========
 * on_gpu_toggled と同じ self-restart 機構 (env セット → save_session → execv)。
 * registry エントリの gpu_mode + force_single (threading 軸) を env にセットして再起動。
 * BLINKGTK_GPU_MODE / BLINKGTK_FORCE_SINGLE_THREADED は起動時のみ読取のため再起動が前提。
 * 経路選択を log marker 化 (Reviewer E: harness が経路別 content_present を correlate)。 */
static void on_render_path_changed(GObject* obj, GParamSpec* pspec,
                                   gpointer user_data) {
  (void)pspec;
  (void)user_data;
  guint sel = gtk_drop_down_get_selected(GTK_DROP_DOWN(obj));
  /* CP-1073 rule 5: dropdown は listable のみ掲載のため map で registry idx に変換。 */
  if (sel >= g_listable_count) {
    return;
  }
  const BlinkRenderPath* p = &g_render_paths[g_listable_map[sel]];
  const char* cur_mode = getenv("BLINKGTK_GPU_MODE");
  const char* cur_single = getenv("BLINKGTK_FORCE_SINGLE_THREADED");
  int cur_single_on = (cur_single && cur_single[0] == '1') ? 1 : 0;
  const char* cur_mojo = getenv("BLINKGTK_USE_MOJO_IPC");
  int cur_mojo_on = (cur_mojo && cur_mojo[0] == '1') ? 1 : 0;
  /* 変化なし → 再起動しない (初期 set 後の no-op 選択も保護)。
   * CP-1071: 標準 software と mojo は同 gpu_mode=software のため、use_mojo 軸を
   * 比較に含めないと切替が no-op 誤判定される。 */
  if (cur_mode && strcmp(cur_mode, p->gpu_mode) == 0 &&
      cur_single_on == p->force_single && cur_mojo_on == p->use_mojo) {
    return;
  }
  setenv("BLINKGTK_GPU_MODE", p->gpu_mode, 1);
  if (p->force_single) {
    setenv("BLINKGTK_FORCE_SINGLE_THREADED", "1", 1);
  } else {
    unsetenv("BLINKGTK_FORCE_SINGLE_THREADED");
  }
  if (p->use_mojo) {
    setenv("BLINKGTK_USE_MOJO_IPC", "1", 1);
  } else {
    unsetenv("BLINKGTK_USE_MOJO_IPC");
  }
  /* log marker (Reviewer E: harness 経路別 correlate 用、id を literal 出力) */
  printf("[CP-1063] 描画経路 → id=%s mode=%s single=%d mojo=%d [%s] %s — "
         "session 保存して execv 再起動\n",
         p->id, p->gpu_mode, p->force_single, p->use_mojo, p->status, p->label);
  fflush(stdout);
  save_session();
  char* new_argv[] = { g_argv0, (char*)"--restore-session", NULL };
  execv("/proc/self/exe", new_argv);
  perror("[CP-1063] execv() failed");
}

/* CP-1073 案A / CP-1074 S: dashboard load 完了時に現経路情報 (env + 長所/短所/
 * 性能スコア) を 1 回 JSON 注入。env はセッション固定のため load 駆動 (周期しない)。
 * JS は single quote、registry 文字列に ' は無い (検証マトリックス準拠)。 */
static void on_inspector_dash_loaded(GtkWidget* widget, gint load_event,
                                     gpointer user_data) {
  (void)user_data;
  if (load_event != 2) return;  /* BLINK_LOAD_FINISHED */
  BlinkWebView* view = BLINK_WEB_VIEW(widget);
  if (!view) return;
  const char* gm = getenv("BLINKGTK_GPU_MODE");
  const char* fs = getenv("BLINKGTK_FORCE_SINGLE_THREADED");
  const char* mj = getenv("BLINKGTK_USE_MOJO_IPC");
  int fs_on = (fs && fs[0] == '1') ? 1 : 0;
  int mj_on = (mj && mj[0] == '1') ? 1 : 0;
  guint cur = 0;
  for (guint i = 0; i < BLINK_RENDER_PATH_COUNT; i++) {
    if (((gm && strcmp(gm, g_render_paths[i].gpu_mode) == 0) ||
         (!gm && strcmp(g_render_paths[i].gpu_mode, "software") == 0)) &&
        fs_on == g_render_paths[i].force_single &&
        mj_on == g_render_paths[i].use_mojo) {
      cur = i;
      break;
    }
  }
  const BlinkRenderPath* p = &g_render_paths[cur];
  char js[1400];
  snprintf(js, sizeof(js),
           "window.blinkgtkInspectorUpdate({pathInfo:{label:'%s',mode:'%s',"
           "mojo:%d,single:%d,perf:'%s',pros:'%s',cons:'%s'}})",
           p->label, gm ? gm : "software", mj_on, fs_on, p->perf, p->pros,
           p->cons);
  blink_web_view_execute_javascript(view, js, NULL, NULL);
  printf("[CP-1074] pathInfo injected (load-changed) id=%s\n", p->id);
  fflush(stdout);
}

/* CP-1073 P1 (6/5 Lead 承認): BlinkGTK インスペクタ右ペイン。仕様
 * (BLINKGTK_INSPECTOR_DESIGN-ja.md) の「案1 = 第二 BlinkWebView で neon

 * toggle 非表示時も hide 保持 (再生成コスト回避)。経路特徴(長所/短所/性能スコア)
 * は dashboard HTML 内「経路情報」セクションに統合予定 (CP-1073 §2 Q1、P2 で JSON 注入)。 */
static GtkWidget* build_inspector_pane(void) {
  GtkWidget* stack = gtk_stack_new();
  GtkWidget* dash = blink_web_view_new();
  g_inspector_dash_view = BLINK_WEB_VIEW(dash);
  /* CP-1074 S: dashboard load 完了時に経路情報 (env + 長所/短所/性能) を 1 回注入。 */
  g_signal_connect(dash, "load-changed",
                   G_CALLBACK(on_inspector_dash_loaded), NULL);

  /* dashboard HTML の URI 解決: binary 同梱 (collocate) 優先 → repo fallback。 */
  char uri[1200];
  uri[0] = '\0';
  char exe[1024];
  ssize_t n = readlink("/proc/self/exe", exe, sizeof(exe) - 1);
  if (n > 0) {
    exe[n] = '\0';
    char* slash = strrchr(exe, '/');
    if (slash) {
      *slash = '\0';
      char cand[1200];
      snprintf(cand, sizeof(cand), "%s/blinkgtk_inspector_dashboard.html", exe);
      if (access(cand, F_OK) == 0) {
        snprintf(uri, sizeof(uri), "file://%s", cand);
      }
    }
  }
  if (uri[0] == '\0') {
    const char* repo = "$BLINKGTK_ROOT/tests/web/"
                       "blinkgtk_inspector_dashboard.html";
    if (access(repo, F_OK) == 0) {
      snprintf(uri, sizeof(uri), "file://%s", repo);
    }
  }
  if (uri[0] != '\0') {
    blink_web_view_load_uri(g_inspector_dash_view, uri);
    printf("[CP-1073 P1] inspector dashboard load: %s\n", uri);
  } else {
    printf("[CP-1073 P1] WARN: dashboard HTML 未発見 (collocate/repo)\n");
  }
  fflush(stdout);

  gtk_stack_add_titled(GTK_STACK(stack), dash, "render_path", "描画経路");

  GtkWidget* frame = gtk_frame_new(NULL);
  gtk_frame_set_child(GTK_FRAME(frame), stack);
  gtk_widget_set_size_request(frame, 280, -1);
  return frame;
}

/* ウィンドウクローズ時のコールバック */
/* Build 731: blink_gtk_shutdown() はRunLoop実行中に呼ばれた場合を自動検知し、
 * 内部でQuit()→RunLoop終了後にシャットダウンを安全に実行する。
 * アプリ側はblink_gtk_shutdown()を呼ぶだけでよい（BlinkGTKライブラリ側が制御）。
 * FALSE = ウィンドウを閉じる（デフォルト処理を継続） */

/*
 * BLINKGTK_ISSUE_84_FIX (CP-963 案 A、2026-05-19、Reviewer A-E 合議承認):
 * Mutter 50.1 unmaximize 後の 2-step sequence 追跡。
 * step 1: notify::maximized で「直前に最大化解除」を記録
 * step 2: notify::is-active で 500ms 以内に inactive 化 → gtk_window_present で再前面化
 *
 * 真因 (Agent 1/2/3 合議結論): Mutter 50.1 は unset_maximized 後に
 *   configure(states=[ACTIVATED]) → configure(states=[]) の 2-step を送る。
 *   step 2 で ACTIVATED が剥奪されウィンドウが背面化。
 *   BlinkGTK は GTK4 経由で xdg_toplevel を所有 (CP-92 path) するため、
 *   Chromium ozone wayland 側 WaylandToplevelWindow の handler は呼ばれない。
 *   → GTK4 GtkWindow signal で捕捉して再前面化する。
 *
 * 無効化: BLINKGTK_MUTTER_REACTIVATE=off
 * 詳細ログ: BLINKGTK_DEBUG_ISSUE_84=1
 */
static gboolean g_was_recently_unmaximized = FALSE;
static gint64 g_unmaximize_time_us = 0;
static int g_issue84_debug = -1;

static int issue84_debug_enabled(void) {
  if (g_issue84_debug < 0) {
    const char* dbg = g_getenv("BLINKGTK_DEBUG_ISSUE_84");
    g_issue84_debug = (dbg && *dbg && g_strcmp0(dbg, "0") != 0) ? 1 : 0;
  }
  return g_issue84_debug;
}

static int issue84_reactivate_enabled(void) {
  const char* env = g_getenv("BLINKGTK_MUTTER_REACTIVATE");
  /* default = on、"off" で完全無効化 */
  return (env && g_strcmp0(env, "off") == 0) ? 0 : 1;
}

static void on_window_maximized_changed(GObject* object,
                                        GParamSpec* pspec,
                                        gpointer user_data) {
  (void)pspec;
  (void)user_data;
  GtkWindow* window = GTK_WINDOW(object);
  gboolean maximized = gtk_window_is_maximized(window);
  if (!maximized) {
    g_was_recently_unmaximized = TRUE;
    g_unmaximize_time_us = g_get_monotonic_time();
    if (issue84_debug_enabled()) {
      g_message("[BLINKGTK_ISSUE_84] notify::maximized = FALSE at %" G_GINT64_FORMAT " us",
                g_unmaximize_time_us);
    }
  } else {
    g_was_recently_unmaximized = FALSE;
  }
}

static void on_window_is_active_changed(GObject* object,
                                        GParamSpec* pspec,
                                        gpointer user_data) {
  (void)pspec;
  (void)user_data;
  GtkWindow* window = GTK_WINDOW(object);
  gboolean is_active = gtk_window_is_active(window);
  if (is_active) {
    g_was_recently_unmaximized = FALSE;
    return;
  }
  if (!g_was_recently_unmaximized) {
    return;
  }
  gint64 now = g_get_monotonic_time();
  gint64 elapsed_ms = (now - g_unmaximize_time_us) / 1000;
  if (elapsed_ms >= 500) {
    /* 500ms 経過後の inactive はユーザー操作と判断、reset */
    g_was_recently_unmaximized = FALSE;
    if (issue84_debug_enabled()) {
      g_message("[BLINKGTK_ISSUE_84] is_active=FALSE elapsed=%" G_GINT64_FORMAT "ms (>= 500ms), ignore",
                elapsed_ms);
    }
    return;
  }
  /* 500ms 以内 → Mutter 50.1 step 2 による背面化と判定 */
  if (!issue84_reactivate_enabled()) {
    if (issue84_debug_enabled()) {
      g_message("[BLINKGTK_ISSUE_84] is_active=FALSE elapsed=%" G_GINT64_FORMAT "ms, "
                "reactivate disabled by BLINKGTK_MUTTER_REACTIVATE=off",
                elapsed_ms);
    }
    g_was_recently_unmaximized = FALSE;
    return;
  }
  g_message("[BLINKGTK_ISSUE_84] is_active=FALSE within %" G_GINT64_FORMAT "ms of unmaximize, "
            "calling gtk_window_present() to recover foreground",
            elapsed_ms);
  gtk_window_present(window);
  g_was_recently_unmaximized = FALSE;
}

static gboolean on_window_close(GtkWindow* window, gpointer user_data) {
  (void)window;
  (void)user_data;
  /* スクロールキャッシュタイマーを停止 */
  if (g_scroll_cache_timer_id > 0) {
    g_source_remove(g_scroll_cache_timer_id);
    g_scroll_cache_timer_id = 0;
  }
  blink_gtk_shutdown();
  return FALSE;
}

   * アプリ側からの明示的なblink_gtk_shutdown()呼び出しは不要。 */
  int ret = blink_gtk_run_main_loop();

  /* 後処理 */
  free(ses_url);
  free(ses_gpu);
  return ret;
}
