/* BlinkGTK message bridge smoke test — Issue #57 回帰検出用最小サンプル
 *
 * blink_web_view_register_message_handler() を呼び、ページをロードして
 * JS↔C bridge script が正常に注入されることを確認する。
 *
 * iter7 以降では、iter6 の libcc.so ICF 不整合由来のクラッシュ
 * (SEGV / GLib handler_unref_R assertion) が発生しないことを保証する。
 *
 * ビルド:
 *   ninja -C out/Release_Component \
 *     third_party/blinkgtk/examples:message_bridge_smoke
 *
 * 実行:
 *   # Weston headless 環境を起動してから
 *   weston --backend=headless-backend.so --width=1280 --height=800 &
 *   export WAYLAND_DISPLAY=wayland-1
 *   ./out/Release_Component/message_bridge_smoke
 *
 * 期待される結果 (12 秒で終了):
 *   - SIGSEGV / SIGABRT / handler_unref_R = 0
 *   - "register_message_handler returned OK" ログ出力
 *   - "[Issue#19] RegisterMessageHandler: name=stats" ログ出力
 *   - BLINK_LOAD_FINISHED 発火
 *
 * 参照: Issue #57 (resolved in iter7), Issue #19 (JS↔C message bridge)
 */

#include <blink_gtk/blink_gtk.h>
#include <gtk/gtk.h>
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>  /* _exit() */

/* JS → C message handler */
static void on_stats(const char* name,
                     const guint8* data,
                     gsize length,
                     gpointer user_data) {
    (void)user_data;
    fprintf(stderr, "[smoke] on_stats: name=%s length=%zu\n",
            name ? name : "(null)", length);
    if (data && length > 0 && length < 256) {
        fprintf(stderr, "[smoke]   data: %.*s\n", (int)length, (const char*)data);
    }
}

/* 12 秒後に終了 (自動化テスト用) */
static gboolean quit_later(gpointer user_data) {
    (void)user_data;
    fprintf(stderr, "[smoke] TIMEOUT_EXIT_OK\n");
    /* AtExitManager teardown を避けるため _exit 使用。
     * 実アプリでは gtk_window_destroy() → 通常の main loop 脱出を使う。
     */
    _exit(0);
    return G_SOURCE_REMOVE;
}

/* 3 秒後に JS 側から postMessage を呼んで on_stats 着火を確認 */
static gboolean trigger_postmessage(gpointer user_data) {
    BlinkWebView* view = (BlinkWebView*)user_data;
    /* JS: window.blinkgtk.postMessage('stats', 'hello from js') */
    blink_web_view_execute_javascript(
        view,
        "if (window.blinkgtk && window.blinkgtk.postMessage) {"
        "  window.blinkgtk.postMessage('stats', 'hello from js');"
        "}");
    fprintf(stderr, "[smoke] Dispatched postMessage from JS\n");
    return G_SOURCE_REMOVE;
}

int main(int argc, char* argv[]) {
    fprintf(stderr, "[smoke] Initializing BlinkGTK\n");
    if (!blink_gtk_init(&argc, &argv)) {
        fprintf(stderr, "[smoke] blink_gtk_init FAILED\n");
        return 1;
    }

    GtkWidget* window = gtk_window_new();
    gtk_window_set_default_size(GTK_WINDOW(window), 1280, 800);
    gtk_window_set_title(GTK_WINDOW(window), "BlinkGTK Message Bridge Smoke");

    BlinkWebView* view = BLINK_WEB_VIEW(blink_web_view_new());
    if (!view) {
        fprintf(stderr, "[smoke] blink_web_view_new FAILED\n");
        return 1;
    }
    gtk_window_set_child(GTK_WINDOW(window), GTK_WIDGET(view));

    /* ★ Issue #57 で SEGV していた API — iter7 以降は正常動作 */
    blink_web_view_register_message_handler(view, "stats", on_stats, NULL);
    fprintf(stderr, "[smoke] register_message_handler returned OK\n");

    gtk_window_present(GTK_WINDOW(window));

    /* シンプルな HTML ページをロード (bridge script 注入トリガ) */
    blink_web_view_load_uri(view,
        "data:text/html,<!DOCTYPE html>"
        "<html><body><h1>BlinkGTK Message Bridge Smoke</h1>"
        "<p>Test page for JS->C message bridge.</p></body></html>");

    /* 6 秒後: JS → C postMessage 送信 */
    g_timeout_add_seconds(6, trigger_postmessage, view);
    /* 12 秒後: 自動終了 */
    g_timeout_add_seconds(12, quit_later, NULL);

    return blink_gtk_run_main_loop();
}
