Overview

BlinkGTK uses the GObject signal system to notify about page load state and navigation events.

What Are GObject Signals

GObject signals are at the core of GTK/GLib event-driven programming:

  • Type-safe: Type checking at compile time
  • Flexible connection: Multiple handlers can be connected to the same signal
  • Automatic disconnection: Signals are automatically disconnected when the object is destroyed

BlinkGTK's Signal Architecture

Chromium/Blink Engine

Basic Usage

#include <blink_gtk/blink_gtk.h>

static void on_load_changed(GtkWidget* web_view,
                            BlinkLoadEvent load_event,
                            gpointer user_data) {
    const char* event_name = "";
    switch (load_event) {
        case BLINK_LOAD_STARTED:
            event_name = "STARTED";
            break;
        case BLINK_LOAD_REDIRECTED:
            event_name = "REDIRECTED";
            break;
        case BLINK_LOAD_COMMITTED:
            event_name = "COMMITTED";
            break;
        case BLINK_LOAD_FINISHED:
            event_name = "FINISHED";
            break;
    }
    g_print("Load event: %s\n", event_name);
}

int main(int argc, char* argv[]) {
    gtk_init();  /* GTK4 は引数なし */
    blink_gtk_init(&argc, &argv);  /* GTK4 は引数なし */

    GtkWidget* web_view = blink_web_view_new();

    // Connect signal
    g_signal_connect(web_view, "load-changed",
                     G_CALLBACK(on_load_changed), NULL);

    blink_web_view_load_uri(BLINK_WEB_VIEW(web_view), "https://example.com");

    // ... (GTK main loop, etc.)

    return 0;
}

Load Events

load-changed

Overview: Emitted when the page load state changes.

Signature

void user_function(GtkWidget* web_view,
                   BlinkLoadEvent load_event,
                   gpointer user_data);

Parameters

Parameter Type Description
web_view GtkWidget* The WebView widget
load_event BlinkLoadEvent The kind of load event
user_data gpointer User data

Values of BlinkLoadEvent

typedef enum {
    BLINK_LOAD_STARTED,      // Navigation started
    BLINK_LOAD_REDIRECTED,   // Redirect occurred
    BLINK_LOAD_COMMITTED,    // Response received, DOM construction started
    BLINK_LOAD_FINISHED      // Page load complete
} BlinkLoadEvent;
Value Meaning Emission timing
BLINK_LOAD_STARTED Load started Immediately after blink_web_view_load_uri()
BLINK_LOAD_REDIRECTED Redirect When an HTTP redirect occurs
BLINK_LOAD_COMMITTED Committed After the response headers are received
BLINK_LOAD_FINISHED Finished DOMContentLoaded + resource loading complete

Example 1: Displaying a Progress Bar

typedef struct {
    GtkWidget* progress_bar;
    GtkWidget* status_label;
} AppData;

static void on_load_changed(GtkWidget* web_view,
                            BlinkLoadEvent load_event,
                            gpointer user_data) {
    AppData* data = (AppData*)user_data;

    switch (load_event) {
        case BLINK_LOAD_STARTED:
            gtk_progress_bar_set_fraction(GTK_PROGRESS_BAR(data->progress_bar), 0.0);
            gtk_label_set_text(GTK_LABEL(data->status_label), "Loading...");
            gtk_widget_show(data->progress_bar);
            break;

        case BLINK_LOAD_REDIRECTED:
            gtk_progress_bar_set_fraction(GTK_PROGRESS_BAR(data->progress_bar), 0.3);
            gtk_label_set_text(GTK_LABEL(data->status_label), "Redirecting...");
            break;

        case BLINK_LOAD_COMMITTED:
            gtk_progress_bar_set_fraction(GTK_PROGRESS_BAR(data->progress_bar), 0.6);
            gtk_label_set_text(GTK_LABEL(data->status_label), "Rendering...");
            break;

        case BLINK_LOAD_FINISHED:
            gtk_progress_bar_set_fraction(GTK_PROGRESS_BAR(data->progress_bar), 1.0);
            gtk_label_set_text(GTK_LABEL(data->status_label), "Done");
            // Hide the progress bar after 1 second
            g_timeout_add(1000, (GSourceFunc)gtk_widget_hide, data->progress_bar);
            break;
    }
}

Example 2: Enabling/Disabling Navigation Buttons

static void on_load_changed(GtkWidget* web_view,
                            BlinkLoadEvent load_event,
                            gpointer user_data) {
    if (load_event != BLINK_LOAD_FINISHED)
        return;

    // Update the back/forward button states after the load completes
    gboolean can_go_back = blink_web_view_can_go_back(BLINK_WEB_VIEW(web_view));
    gboolean can_go_forward = blink_web_view_can_go_forward(BLINK_WEB_VIEW(web_view));

    gtk_widget_set_sensitive(back_button, can_go_back);
    gtk_widget_set_sensitive(forward_button, can_go_forward);
}

load-failed

Overview: Emitted when the page load fails.

Signature

void user_function(GtkWidget* web_view,
                   const char* failing_url,
                   int error_code,
                   gpointer user_data);

Parameters

Parameter Type Description
web_view GtkWidget* The WebView widget
failing_url const char* The URL that failed
error_code int Chromium error code (negative value)
user_data gpointer User data

Main Error Codes

Error code Constant name (reference) Meaning
-2 net::ERR_FAILED General error
-3 net::ERR_ABORTED Load aborted
-7 net::ERR_TIMED_OUT Timeout
-21 net::ERR_NETWORK_CHANGED Network changed
-105 net::ERR_NAME_NOT_RESOLVED DNS resolution failed
-106 net::ERR_INTERNET_DISCONNECTED Internet not connected
-324 net::ERR_EMPTY_RESPONSE Empty response

Example 1: Displaying an Error Message

static void on_load_failed(GtkWidget* web_view,
                          const char* failing_url,
                          int error_code,
                          gpointer user_data) {
    const char* error_message = "Unknown error";

    // Message based on the error code
    switch (error_code) {
        case -2:
            error_message = "Failed to load the page";
            break;
        case -3:
            error_message = "Load was aborted";
            break;
        case -7:
            error_message = "The connection timed out";
            break;
        case -105:
            error_message = "The server could not be found";
            break;
        case -106:
            error_message = "Not connected to the Internet";
            break;
    }

    // Show an error dialog
    GtkWidget* dialog = gtk_message_dialog_new(
        GTK_WINDOW(gtk_widget_get_root(web_view)),
        GTK_DIALOG_MODAL,
        GTK_MESSAGE_ERROR,
        GTK_BUTTONS_OK,
        "Load error"
    );
    gtk_message_dialog_format_secondary_text(
        GTK_MESSAGE_DIALOG(dialog),
        "%s\n\nURL: %s\nError code: %d",
        error_message, failing_url, error_code
    );

    g_signal_connect(dialog, "response", G_CALLBACK(gtk_window_destroy), NULL);
    gtk_widget_show(dialog);
}

Navigation Events

title-changed

Overview: Emitted when the page title changes.

Signature

void user_function(GtkWidget* web_view,
                   const char* title,
                   gpointer user_data);

Parameters

Parameter Type Description
web_view GtkWidget* The WebView widget
title const char* The new page title (UTF-8)
user_data gpointer User data

Emission Timing

  • When page loading completes (the content of the <title> tag)
  • When document.title is changed via JavaScript
  • When a page transition occurs in an SPA

Example 1: Updating the Window Title

static void on_title_changed(GtkWidget* web_view,
                            const char* title,
                            gpointer user_data) {
    GtkWindow* window = GTK_WINDOW(user_data);

    // "Title - App name" format
    char* window_title = g_strdup_printf("%s - MyBrowser", title);
    gtk_window_set_title(window, window_title);
    g_free(window_title);
}

// Connect the signal (pass window as user_data)
g_signal_connect(web_view, "title-changed",
                 G_CALLBACK(on_title_changed), window);

Example 2: Updating a Tab Title

typedef struct {
    GtkWidget* tab_label;
    GtkWidget* web_view;
} TabData;

static void on_title_changed(GtkWidget* web_view,
                            const char* title,
                            gpointer user_data) {
    TabData* tab = (TabData*)user_data;

    // Truncate the title if it is too long
    char* display_title;
    if (strlen(title) > 30) {
        display_title = g_strdup_printf("%.27s...", title);
    } else {
        display_title = g_strdup(title);
    }

    gtk_label_set_text(GTK_LABEL(tab->tab_label), display_title);
    g_free(display_title);
}

BlinkGTK WebKitGTK Compatibility

Note: WebKitGTK uses the notify::title property change notification, whereas BlinkGTK provides a dedicated title-changed signal.

uri-changed

Overview: Emitted when the page URI changes.

Signature

void user_function(GtkWidget* web_view,
                   const char* uri,
                   gpointer user_data);

Parameters

Parameter Type Description
web_view GtkWidget* The WebView widget
uri const char* The new URI (UTF-8)
user_data gpointer User data

Emission Timing

  • When URL loading starts via blink_web_view_load_uri()
  • When navigating via blink_web_view_go_back() / go_forward()
  • When an in-page link is clicked
  • When location.href is changed via JavaScript
  • When the History API is used in an SPA

Example 1: Updating the Address Bar

static void on_uri_changed(GtkWidget* web_view,
                          const char* uri,
                          gpointer user_data) {
    GtkEntry* url_entry = GTK_ENTRY(user_data);

    // Show the URI in the address bar
    gtk_editable_set_text(GTK_EDITABLE(url_entry), uri);
}

// Connect the signal (pass url_entry as user_data)
g_signal_connect(web_view, "uri-changed",
                 G_CALLBACK(on_uri_changed), url_entry);

Example 2: History Management

typedef struct {
    GList* history;      // History list
    int current_index;   // Current position
} HistoryManager;

static void on_uri_changed(GtkWidget* web_view,
                          const char* uri,
                          gpointer user_data) {
    HistoryManager* history = (HistoryManager*)user_data;

    // Remove history entries after the current position (new branch)
    while (history->current_index < g_list_length(history->history) - 1) {
        GList* last = g_list_last(history->history);
        g_free(last->data);
        history->history = g_list_delete_link(history->history, last);
    }

    // Append the new URI to the history
    history->history = g_list_append(history->history, g_strdup(uri));
    history->current_index = g_list_length(history->history) - 1;

    g_print("History: %d items, current=%d\n",
            g_list_length(history->history),
            history->current_index);
}

Example 3: Displaying a Secure Page Indicator

static void on_uri_changed(GtkWidget* web_view,
                          const char* uri,
                          gpointer user_data) {
    GtkImage* security_icon = GTK_IMAGE(user_data);

    // Check whether the page is served over HTTPS
    if (g_str_has_prefix(uri, "https://")) {
        gtk_image_set_from_icon_name(security_icon, "dialog-password");
        gtk_widget_set_tooltip_text(GTK_WIDGET(security_icon),
                                    "This page is secure");
    } else {
        gtk_image_set_from_icon_name(security_icon, "dialog-warning");
        gtk_widget_set_tooltip_text(GTK_WIDGET(security_icon),
                                    "This page is not secure");
    }
}

Practical Patterns

Pattern 1: Complete Browser UI Update

An implementation example of a complete browser UI that combines multiple signals.

#include <gtk/gtk.h>
#include <blink_gtk/blink_gtk.h>

typedef struct {
    GtkWidget* window;
    GtkWidget* url_entry;
    GtkWidget* back_button;
    GtkWidget* forward_button;
    GtkWidget* reload_button;
    GtkWidget* stop_button;
    GtkWidget* progress_bar;
    GtkWidget* status_label;
    GtkWidget* security_icon;
    GtkWidget* web_view;
} BrowserUI;

/* load-changed: update the UI based on load state */
static void on_load_changed(GtkWidget* web_view,
                            BlinkLoadEvent load_event,
                            gpointer user_data) {
    BrowserUI* ui = (BrowserUI*)user_data;

    switch (load_event) {
        case BLINK_LOAD_STARTED:
            gtk_progress_bar_set_fraction(GTK_PROGRESS_BAR(ui->progress_bar), 0.0);
            gtk_widget_set_visible(ui->progress_bar, TRUE);
            gtk_widget_set_visible(ui->stop_button, TRUE);
            gtk_widget_set_visible(ui->reload_button, FALSE);
            gtk_label_set_text(GTK_LABEL(ui->status_label), "Loading...");
            break;

        case BLINK_LOAD_COMMITTED:
            gtk_progress_bar_set_fraction(GTK_PROGRESS_BAR(ui->progress_bar), 0.5);
            break;

        case BLINK_LOAD_FINISHED: {
            gtk_progress_bar_set_fraction(GTK_PROGRESS_BAR(ui->progress_bar), 1.0);
            gtk_widget_set_visible(ui->progress_bar, FALSE);
            gtk_widget_set_visible(ui->stop_button, FALSE);
            gtk_widget_set_visible(ui->reload_button, TRUE);
            gtk_label_set_text(GTK_LABEL(ui->status_label), "Done");

            // Update the navigation button states
            gboolean can_go_back = blink_web_view_can_go_back(BLINK_WEB_VIEW(web_view));
            gboolean can_go_forward = blink_web_view_can_go_forward(BLINK_WEB_VIEW(web_view));
            gtk_widget_set_sensitive(ui->back_button, can_go_back);
            gtk_widget_set_sensitive(ui->forward_button, can_go_forward);
            break;
        }
    }
}

/* load-failed: show the error */
static void on_load_failed(GtkWidget* web_view,
                          const char* failing_url,
                          int error_code,
                          gpointer user_data) {
    BrowserUI* ui = (BrowserUI*)user_data;

    char* message = g_strdup_printf("Load error (code: %d)", error_code);
    gtk_label_set_text(GTK_LABEL(ui->status_label), message);
    g_free(message);

    gtk_widget_set_visible(ui->progress_bar, FALSE);
    gtk_widget_set_visible(ui->stop_button, FALSE);
    gtk_widget_set_visible(ui->reload_button, TRUE);
}

/* title-changed: update the window title */
static void on_title_changed(GtkWidget* web_view,
                            const char* title,
                            gpointer user_data) {
    BrowserUI* ui = (BrowserUI*)user_data;

    char* window_title = g_strdup_printf("%s - MyBrowser", title);
    gtk_window_set_title(GTK_WINDOW(ui->window), window_title);
    g_free(window_title);
}

/* uri-changed: update the address bar and security icon */
static void on_uri_changed(GtkWidget* web_view,
                          const char* uri,
                          gpointer user_data) {
    BrowserUI* ui = (BrowserUI*)user_data;

    // Update the address bar
    gtk_editable_set_text(GTK_EDITABLE(ui->url_entry), uri);

    // Update the security icon
    if (g_str_has_prefix(uri, "https://")) {
        gtk_image_set_from_icon_name(GTK_IMAGE(ui->security_icon),
                                    "dialog-password");
    } else {
        gtk_image_set_from_icon_name(GTK_IMAGE(ui->security_icon),
                                    "dialog-warning");
    }
}

/* Connect the signals */
static void setup_signals(BrowserUI* ui) {
    g_signal_connect(ui->web_view, "load-changed",
                     G_CALLBACK(on_load_changed), ui);
    g_signal_connect(ui->web_view, "load-failed",
                     G_CALLBACK(on_load_failed), ui);
    g_signal_connect(ui->web_view, "title-changed",
                     G_CALLBACK(on_title_changed), ui);
    g_signal_connect(ui->web_view, "uri-changed",
                     G_CALLBACK(on_uri_changed), ui);
}

Pattern 2: Blocking/Unblocking Signals

An example of how to temporarily disable a signal.

static void load_url_without_history(GtkWidget* web_view,
                                     const char* url,
                                     gpointer handler_id) {
    // Temporarily block the uri-changed signal
    g_signal_handler_block(web_view, GPOINTER_TO_UINT(handler_id));

    // Load the URL
    blink_web_view_load_uri(BLINK_WEB_VIEW(web_view), url);

    // Unblock the signal
    g_signal_handler_unblock(web_view, GPOINTER_TO_UINT(handler_id));
}

/* Usage example */
gulong handler_id = g_signal_connect(web_view, "uri-changed",
                                     G_CALLBACK(on_uri_changed), NULL);

// Load a URL without recording it in the history
load_url_without_history(web_view, "https://example.com",
                         GUINT_TO_POINTER(handler_id));

Pattern 3: Disconnecting Signal Handlers

typedef struct {
    gulong load_changed_id;
    gulong load_failed_id;
    gulong title_changed_id;
    gulong uri_changed_id;
} SignalHandlers;

static void disconnect_all_signals(GtkWidget* web_view,
                                   SignalHandlers* handlers) {
    if (handlers->load_changed_id > 0) {
        g_signal_handler_disconnect(web_view, handlers->load_changed_id);
        handlers->load_changed_id = 0;
    }
    if (handlers->load_failed_id > 0) {
        g_signal_handler_disconnect(web_view, handlers->load_failed_id);
        handlers->load_failed_id = 0;
    }
    if (handlers->title_changed_id > 0) {
        g_signal_handler_disconnect(web_view, handlers->title_changed_id);
        handlers->title_changed_id = 0;
    }
    if (handlers->uri_changed_id > 0) {
        g_signal_handler_disconnect(web_view, handlers->uri_changed_id);
        handlers->uri_changed_id = 0;
    }
}

Troubleshooting

Signal Does Not Fire

Symptom: The signal handler is not called

Causes and remedies:

  1. Typo in the signal name

    // No Wrong
g_signal_connect(web_view, "load_changed", ...);  // underscore

// Yes Correct
g_signal_connect(web_view, "load-changed", ...);  // hyphen

  2. Connecting before the WebView is created

    // No Wrong
g_signal_connect(web_view, "load-changed", ...);
web_view = blink_web_view_new();

// Yes Correct
web_view = blink_web_view_new();
g_signal_connect(web_view, "load-changed", ...);

  3. The GTK main loop is not running

    blink_web_view_load_uri(BLINK_WEB_VIEW(web_view), "https://example.com");
// No The signal will not fire (no main loop)

gtk_window_present(GTK_WINDOW(window));
// Yes Signals fire once the main loop is running

Crash in a Signal Handler

Symptom: Segmentation fault inside a signal handler

Causes and remedies:

  1. user_data is NULL

    static void on_load_changed(GtkWidget* web_view,
                            BlinkLoadEvent load_event,
                            gpointer user_data) {
    MyData* data = (MyData*)user_data;
    data->count++;  // No Crashes if user_data is NULL
}

// Remedy: NULL check
if (user_data) {
    MyData* data = (MyData*)user_data;
    data->count++;
}

  2. String lifetime

    static const char* saved_title = NULL;

static void on_title_changed(GtkWidget* web_view,
                            const char* title,
                            gpointer user_data) {
    saved_title = title;  // No title is a temporary pointer
}

// Remedy: copy the string
g_free(saved_title);
saved_title = g_strdup(title);  // Yes Store a copy

Memory Leaks

Symptom: Memory usage keeps increasing

Causes and remedies:

  1. Signal handler is not disconnected

    // Yes Save the handler ID
gulong handler_id = g_signal_connect(web_view, "load-changed",
                                     G_CALLBACK(on_load_changed), NULL);

// Disconnect before destroying the WebView
g_signal_handler_disconnect(web_view, handler_id);

  2. String leak

    gchar* blink_web_view_get_user_agent(BlinkWebView* web_view);
// ↑ This function returns a g_strdup()'d string

// No Memory leak
const char* ua = blink_web_view_get_user_agent(web_view);

// Yes Free it after use
char* ua = blink_web_view_get_user_agent(web_view);
g_print("UA: %s\n", ua);
g_free(ua);
